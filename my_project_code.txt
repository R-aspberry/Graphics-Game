#include "Enemy.h"

Enemy::Enemy(int health, int damage, float speed, float attackInterval, int lane, glm::vec2 position)
    : health(health), damage(damage), speed(speed), attackInterval(attackInterval),timeSinceLastAttack(attackInterval), lane(lane), maxHealth(health), position(position), state(EnemyState::MOVING) , MovementBlocked(false) {}

void Enemy:: update(float deltaTime) {
    if (deltaTime < 0.0f) {
        return;
    }
    if (state == EnemyState::MOVING && !MovementBlocked)
        moveTowardsBase(deltaTime);
    if (state != EnemyState::DEAD)
        updateAttackTimer(deltaTime);
}

void Enemy::moveTowardsBase(float deltaTime) {
    position.x = position.x - speed * deltaTime;
}

void Enemy::takeDamage(int incomingDamage){
    if (state == EnemyState::DEAD)
        return;
    if (incomingDamage > 0){
        health -= incomingDamage;
        if (health <= 0){
            health = 0;
            state = EnemyState::DEAD;
        }
    }
}

void Enemy::updateAttackTimer(float deltaTime) {
    if (state == EnemyState::DEAD)  return;
    timeSinceLastAttack += deltaTime;  
}

bool Enemy::canAttack() const {
    if(state == EnemyState::DEAD) return false;
    if(timeSinceLastAttack >= attackInterval)
    {
        return true;
    }
    return false;
}

void Enemy::resetAttackTimer() {
    timeSinceLastAttack = 0.0f;
}

void Enemy::setState(EnemyState newState) {
    if (state == EnemyState::DEAD|| newState == EnemyState::DEAD) return; 
    state = newState;
}

void Enemy::setMovementBlocked(bool blocked){
    MovementBlocked= blocked;
}


bool Enemy::isMovementBlocked() const{
    return MovementBlocked;
}

        MeleePlayer::MeleePlayer(float x, float y, float width, float height, int laneIndex,  int maxHealth, float moveSpeed) : Player (x,y,width,height,laneIndex,maxHealth,moveSpeed){
        }

        MeleePlayer::~MeleePlayer() = default;

        PlayerType MeleePlayer::getPlayerType() const {
            return PlayerType::Melee;
        }

        bool MeleePlayer::canAttack() const {
            if (!(attackCoolDownTimer<=0.0f)){
                return false;
            }
            return actionState == ActionState::Moving;
        }

        void MeleePlayer::startAttack(GameManager &gameManager ) {
            if(!canAttack()){
                return;
            }
            actionState = ActionState::Attacking;
            attackTimer = 0.0f;
            

        }

        void MeleePlayer::finishAttack(){
            if(actionState == ActionState::Attacking){
                actionState = ActionState::Moving;
                attackTimer = 0.0f;
                attackCoolDownTimer = GameConfig::MELEE_ATTACK_INTERVAL;
            }
        }

        int MeleePlayer::getAttackDamage () const{
            return attackDamage;
        }

        float MeleePlayer::getAttackRange() const{
            return attackRange;
        }

        bool MeleePlayer::isAttacking () const{
            return ActionState::Attacking == actionState;
        }

        void MeleePlayer::updateAttack(float deltaTime){
            if(actionState != ActionState::Attacking){
                return;
            }
            attackTimer +=deltaTime;
            if (attackTimer >= Constants::attackDuration){
                finishAttack();
            }
            
        }

        void MeleePlayer::update(float deltaTime){
            Player::update(deltaTime);
            this->updateAttack(deltaTime);
            if (attackCoolDownTimer > 0){
                attackCoolDownTimer-= deltaTime;
            }
            if (attackCoolDownTimer < 0){
                attackCoolDownTimer = 0;
            }
        }

    Player::Player(float x, float y, float width, float height, int laneIndex, int maxHealth, float moveSpeed) : Entity(x, y, width, height, laneIndex) , health(maxHealth),
      maxHealth(maxHealth),
      moveSpeed(moveSpeed),
      actionState(ActionState::Idle),
      PrevactionState(ActionState::Moving),
      movementBlocked(false),
      jumpTimer(0.0f),
      duckTimer(0.0f),
      damageFlashTimer(0.0f),
      attackRange(0.0f){
    }

    Player::~Player()  {

    }

     void Player::takeDamage(int damage){
        if (actionState == ActionState::Defeated){
            return;
        }
        if (damage<=0){
            return ;
        }
        PrevactionState = actionState;
        health-= damage;
        damageFlashTimer = 0.2f;

        if(health <=0){
            health =0;
            setActive(false);
            actionState = ActionState::Defeated;
        }
        else{
            actionState = ActionState::Damaged;
        }
    }

    bool Player::isAlive() const {
        return actionState != ActionState::Defeated /*|| health!=0*/;
    }
    int Player::getHealth() const {
        return health;
    }
    int Player::getMaxHealth() const{
        return maxHealth;
    }
    float Player::getMoveSpeed() const	{
        return moveSpeed;
    }
    ActionState Player::getActionState() const	{
        return actionState;
    }
    void Player::setMovementBlocked(bool blocked)	{
        movementBlocked = blocked;
        //actionState = ActionState::
    }
    bool Player::isMovementBlocked() const	{
        return movementBlocked;
    }
     void Player::jump()	{
        if (actionState == ActionState::Idle || actionState == ActionState::Moving){
            setActionState(ActionState::Jumping);
            jumpTimer = 0.0f;
        }
        else {
            return;
        }
    }
     void Player::duck()	{
        if (actionState == ActionState::Idle || actionState == ActionState::Moving){
            setActionState(ActionState::Ducking);
            duckTimer = 0.0f;
        }
        else {
            return;
        }
    }
    bool Player::isJumping() const	{
        return actionState == ActionState::Jumping;
    }
    bool Player::isDucking() const	{
            return actionState == ActionState::Ducking;
    }

    void Player::updateMovement(float deltaTime) {
        if(ActionState::Moving == actionState && movementBlocked== false){
            x+= moveSpeed * deltaTime;
        }
    }

    void Player::updateJump (float deltaTime){
        if (actionState != ActionState::Jumping){
            return;
        }
        jumpTimer += deltaTime;
        if(jumpTimer >= Constants::JUMP_DURATION){
            jumpTimer = 0.0f;
            actionState = ActionState::Moving;
        }
    }

    void Player::updateDuck (float deltaTime){
        if (actionState != ActionState::Ducking){
            return;
        }
        duckTimer += deltaTime;
        if(duckTimer >=Constants::DUCK_DURATION){
            duckTimer = 0.0f;
            actionState = ActionState::Moving;
        }
    }

    void Player::updateDamageFeedback(float deltaTime)
    {
        if (damageFlashTimer <= 0.0f)
        {
            return;
        }
        damageFlashTimer -= deltaTime;

        if (damageFlashTimer <= 0.0f)
        { damageFlashTimer = 0.0f;
            if(actionState!= ActionState::Defeated){
            actionState = PrevactionState;
        }
        }

        
    }
    void Player::update(float deltaTime)  {
        updateMovement(deltaTime);
        updateJump(deltaTime);
        updateDuck(deltaTime);
        updateDamageFeedback(deltaTime);
    }

    void Player::setActionState(ActionState newState){
        actionState = newState;
    }
     bool Player::canMove() const{
        return !movementBlocked;
    }

    RangedPlayer::RangedPlayer(float x, float y, float width, float height, int laneIndex, int maxHealth, float moveSpeed) : Player (x,y,width,height,laneIndex,maxHealth,moveSpeed){
        

    }

    RangedPlayer::~RangedPlayer() = default;

    PlayerType RangedPlayer::getPlayerType() const {
        return PlayerType::Ranged;
    }

    bool RangedPlayer::canAttack() const {
        if (!(attackCoolDownTimer<=0.0f)){
            return false;
        }
        return actionState == ActionState::Moving;
    }

    void RangedPlayer::startAttack(GameManager& gameManager) {
        if(!canAttack()){
            return;
        }
        actionState = ActionState::Attacking;
        attackTimer = 0.0f;
        auto projectile  = std::make_unique<Projectile>( x + width, y + height / 2.0f - Constants::ProjectileHeight / 2.0f,Constants::ProjectileWidth, Constants::ProjectileHeight,laneIndex ,Constants::ProjectileSpeed,this );
        gameManager.addProjectile(std::move(projectile));
    }

    void RangedPlayer::finishAttack(){
        if(actionState == ActionState::Attacking){
            actionState = ActionState::Moving;
            attackTimer = 0.0f;
            attackCoolDownTimer = Constants::attackIntervals;
        }
    }

    int RangedPlayer::getAttackDamage() const{
        return attackDamage;
    }

    float RangedPlayer::getAttackRange() const{
        return attackRange;
    }

    bool RangedPlayer::isAttacking () const{
        return ActionState::Attacking == actionState;
    }

    void RangedPlayer::updateAttack(float deltaTime){
        if(actionState != ActionState::Attacking){
            return;
        }
        attackTimer +=deltaTime;
        if (attackTimer >= Constants::attackDuration){
            finishAttack();
        }
        
    }

    void RangedPlayer::update(float deltaTime){
        this->updateAttack(deltaTime);
        Player::update(deltaTime);
        attackCoolDownTimer-= deltaTime;
    }

GLuint Game::loadTexture(const char* path){
    int width, height, channels;
    unsigned char* data = stbi_load(path, &width,&height,&channels,4);
    // if (data == nullptr){
    //     return 0;
    // }


    GLuint textureID =0 ;
    glGenTextures(1,&textureID);
    glBindTexture( GL_TEXTURE_2D, textureID);

    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);


    glTexImage2D(GL_TEXTURE_2D,0,GL_RGBA,width,height,0,GL_RGBA,GL_UNSIGNED_BYTE,data);
    stbi_image_free(data);
    return textureID;

}

bool Game::init(){
    if (shader.Program == 0){
        return false;
    }

    rendere = std::make_unique<Renderer>(shader, 1000.0f, 700.0f);

    meleeTexture = loadTexture("assets/sprites/Melee_M1.png");
    defenseTexture = loadTexture("assets/sprites/Defense.png");
    rangedTexture = loadTexture("assets/sprites/Ranged_M1.png");
    
    if(meleeTexture == 0 || rangedTexture ==0 || defenseTexture == 0){
        return false;
    }

    initialized = true;
    return true;

}

void Game::renderPlayers()
{
    const auto& players = gameManager.getActivePlayers();

    for(const auto& player : players){
        GLuint texture =0;

        if(player->getPlayerType() == PlayerType::Defense){
            texture = defenseTexture;
        }
        if(player->getPlayerType() == PlayerType::Melee){
            texture = meleeTexture;
        }
        if(player->getPlayerType() == PlayerType::Ranged){
            texture = rangedTexture;
        }

        rendere->drawSprite(texture, player->getX(), WorldLayout::toRenderY(player.getY(),player.getHeight()), player->getWidth(), player->getHeight());
    }

}

void Game::render() {

    if (!initialized || rendere == nullptr) {
        return;
    }

    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
    renderPlayers();
}

void Game::update(float deltaTime){
    gameManager.update(deltaTime);
    waveManager.update(deltaTime);
}

static const char *vertexShaderSource = R"(
    #version 330 core

    layout(location = 0) in vec3 aPos;
    layout(location = 1) in vec2 aTexCoord;

    out vec2 TexCoord;

    void main() {
        gl_Position = vec4(aPos, 1.0);
        TexCoord = aTexCoord;
    }
    )";

static const char *fragmentShaderSource = R"(
    #version 330 core

    in vec2 TexCoord;
    out vec4 FragColor;

    uniform sampler2D backgroundTexture;
    uniform vec4 laneBounds[4];
    uniform vec4 selectedFlags;
    uniform int hoveredLane;

    void main() {
        vec4 color = texture(backgroundTexture, TexCoord);

        float x = TexCoord.x;
        float y = 1.0 - TexCoord.y;

        for (int i = 0; i < 4; i++) {
            vec4 bounds = laneBounds[i];

            if (x >= bounds.x &&
                x <= bounds.y &&
                y >= bounds.z &&
                y <= bounds.w) {

                if (selectedFlags[i] > 0.5) {
                    color.rgb = mix(
                        color.rgb,
                        vec3(0.6, 0.2, 1.0),
                        0.4
                    );
                }
                else if (hoveredLane == i) {
                    color.rgb = mix(
                        color.rgb,
                        vec3(0.0, 0.9, 1.0),
                        0.3
                    );
                }
            }
        }

        FragColor = color;
    }
    )";

static GLuint compileShader(GLenum kind, const char *source)
{
    GLuint shader = glCreateShader(kind);

    glShaderSource(shader, 1, &source, nullptr);
    glCompileShader(shader);

    GLint success = 0;
    glGetShaderiv(shader, GL_COMPILE_STATUS, &success);

    if (!success)
    {
        char log[1024] = {};

        glGetShaderInfoLog(
            shader,
            sizeof(log),
            nullptr,
            log);

        std::cerr << "Shader error: " << log << '\n';

        glDeleteShader(shader);
        return 0;
    }

    return shader;
}

void mouseButtonCallback(
    GLFWwindow *window,
    int button,
    int action,
    int mods)
{
    (void)mods;

    if (button != GLFW_MOUSE_BUTTON_LEFT ||
        action != GLFW_PRESS)
    {
        return;
    }

    if (currentScreen != AppScreen::MENU)
    {
        return;
    }

    double mouseX = 0.0;
    double mouseY = 0.0;

    glfwGetCursorPos(window, &mouseX, &mouseY);

    int width = 0;
    int height = 0;

    glfwGetWindowSize(window, &width, &height);

    if (width <= 0 || height <= 0)
    {
        return;
    }

    double x = (2.0 * mouseX / width) - 1.0;
    double y = 1.0 - (2.0 * mouseY / height);

    // START BUTTON

    if (x >= -0.46 && x <= -0.10 &&
        y >= 0.00 && y <= 0.20)
    {

        // Go directly to gameplay for now.
        // CUTSCENE can be connected later.

        currentScreen = AppScreen::PLAYING;
        suppressGameplayClick = true;

        std::cout << "Start clicked!\n";
    }

    // EXIT BUTTON

    else if (x >= 0.10 && x <= 0.46 &&
             y >= 0.00 && y <= 0.20)
    {

        glfwSetWindowShouldClose(window, GLFW_TRUE);
    }
}

#include "Shader.h"


Shader::Shader (const char* vertex, const char* fragment){
    GLuint vertexID = glCreateShader(GL_VERTEX_SHADER);
    glShaderSource(vertexID ,1,  &vertex, nullptr);
    glCompileShader(vertexID);
    int success;
    glGetShaderiv(vertexID, GL_COMPILE_STATUS, &success);

    if(!success){
        glDeleteShader(vertexID);
        return ;
    }

    GLuint fragmentID = glCreateShader(GL_FRAGMENT_SHADER);
    glShaderSource(fragmentID ,1,  &fragment, nullptr);
    glCompileShader(fragmentID);
    glGetShaderiv(fragmentID, GL_COMPILE_STATUS, &success);

    if(!success){
        glDeleteShader(fragmentID);
        glDeleteShader(vertexID);
        return ;
    }

    Program = glCreateProgram();

    glAttachShader(Program, vertexID);
    glAttachShader(Program, fragmentID);

    glLinkProgram(Program);

    glGetProgramiv(Program, GL_LINK_STATUS, &success);

    if(!success){
        glDeleteProgram(Program);
        Program = 0;
       
    }

    glDeleteShader(vertexID);
    glDeleteShader(fragmentID);



}

void Shader::use(){
    if(Program != 0){
        glUseProgram(Program);
    }
}

    bool CollisionSystem::intersects(const Rect& a, const Rect& b) const{
        return (a.x < b.x + b.width) && (a.x + a.width > b.x) &&
               (a.y < b.y + b.height) && (a.y + a.height > b.y);
    }

    bool CollisionSystem::intersects( Entity& a,  Entity& b) const{
        return intersects(a.getBounds(), b.getBounds());
    }

    bool CollisionSystem::projectileHitsTarget( Projectile& projectile,  Entity& target) const{
        return intersects(projectile.getBounds(), target.getBounds());
    }
    
    bool CollisionSystem::sameLane( Entity& a,  Entity& b) const{
        return a.getLaneIndex() == b.getLaneIndex();
    }

    bool CollisionSystem::playerEnemyEncounter( Player& player,  Enemy& enemy) const{ 
       if(player.getLaneIndex()!= enemy.getLane()){
        return false;
       }
       Rect enemyBounds = {
            enemy.getPosition().x,
            enemy.getPosition().y,
            WorldLayout::ENEMY_WIDTH,
            WorldLayout::ENEMY_HEIGHT
        };

        Rect playerBounds = player.getBounds();

       if(intersects(playerBounds, enemyBounds)){
            player.setMovementBlocked(true);
            enemy.setMovementBlocked(true);
            return true;
       }
       return false;
    }

void CombatSystem::resolvePlayerAttacks(const std::vector<Enemy*>& activeEnemies){
    std::vector<Player*> activePlayers = (*playerManager).getActivePlayers();

    for (int i=0 ; i<activePlayers.size();i++){
        Player* player= activePlayers[i];
        if ((*player).canAttack()){
            for(int j=0;j<activeEnemies.size();j++){
                Enemy* enemy = activeEnemies[j];
                if (enemy!=nullptr && (*enemy).getHealth() > 0 &&(*collisionSystem).playerEnemyEncounter(*player, *enemy)){
                    resolvePlayerAttack(*player, *enemy);
                    break;}
            }

        }
    }
}

void CombatSystem::resolvePlayerAttack(Player& player, Enemy& enemy){
    player.startAttack(*playerManager);
}

void CombatSystem::resolveEnemyAttack(Enemy& enemy, Player& player)
{
    if (enemy.canAttack() && !isPlayerProtected(player, enemies)){
        player.takeDamage(enemy.getDamage());
        enemy.resetAttackTimer();
    }
}

bool CombatSystem::isPlayerProtected(
    const Player& player,
    const std::vector<Enemy*>&) const
{
    if (playerManager == nullptr) {
        return false;
    }

    const std::vector<Player*> activePlayers = playerManager->getActivePlayers();
    return std::any_of(activePlayers.begin(), activePlayers.end(),
        [&player](const Player* candidate) {
            const auto* defensePlayer =
                dynamic_cast<const DefensePlayer*>(candidate);
            return defensePlayer != nullptr && defensePlayer->protects(player);
        });
}

void CombatSystem::resolveEnemyAttacks(const std::vector<Enemy*>& activeEnemies){
    std::vector<Player*> activePlayers = (*playerManager).getActivePlayers();
    for (int i=0;i<activeEnemies.size();i++){
        Enemy* enemy = activeEnemies[i];
        if (enemy !=nullptr && (*enemy).getHealth() > 0){
            AttackType type = (*enemy).getAttackType();
            for(int j=0;j<activePlayers.size();j++){
                Player* player = activePlayers[j];
                if (player!=nullptr && (*player).isAlive()&& (*player).getLaneIndex() == (*enemy).getLane()){
                    float distance= (*enemy).getPosition().x - (*player).getX();
                    if (type == AttackType::Melee)
                    {
                        if (distance >= 0 && distance <= (*player).getWidth())
                        {
                            (*enemy).setMovementBlocked(true);
                            (*player).setMovementBlocked(true);

                            if ((*enemy).canAttack())
                            {
                                resolveEnemyAttack(*enemy, *player);
                            }

                            break;
                        }
                    }
                    
                    else if (type == AttackType::High)
                    {
                        if (!(*player).isDucking())
                        {
                            resolveEnemyAttack(*enemy, *player);
                            break;
                        }
                    }

                    else if (type== AttackType::Low){
                        if(!(*player).isJumping()){
                            resolveEnemyAttack(*enemy, *player);
                            break;
                        }
                    }
                }
            }
        }
    }

    
}

                totalEnemiesInWave(GameConfig::WAVE_1_COUNT), spawnTimer(0.0f),gameWon(false) {}


void WaveManager::processEnemyDeaths() {
    for (std::size_t i = 0; i < activeEnemies.size(); ++i) {
        if(activeEnemies[i]->getState() == EnemyState::DEAD && !dropGenerated[i]){
            int reward = GameConfig::EASY_ENEMY_REWARD;  
            if(activeEnemies[i]->getEnemyType() == EnemyType::EASY){
                reward = GameConfig::EASY_ENEMY_REWARD;
            }
            else if(activeEnemies[i]->getEnemyType() == EnemyType::MEDIUM){
                reward = GameConfig::MEDIUM_ENEMY_REWARD;
            }
            else if(activeEnemies[i]->getEnemyType() == EnemyType::HARD){
                reward = GameConfig::HARD_ENEMY_REWARD;
            }
            glm::vec2 enemyPosition = activeEnemies[i]->getPosition();

            glm::vec2 dropPosition(
                enemyPosition.x + (WorldLayout::ENEMY_WIDTH - WorldLayout::COLLECTIBLE_WIDTH) / 2.0f,

                enemyPosition.y + (WorldLayout::ENEMY_HEIGHT - WorldLayout::COLLECTIBLE_HEIGHT) / 2.0f
            );

            collectibles.push_back({
                activeEnemies[i]->getLane(),
                dropPosition,
                reward
            });
            dropGenerated[i] = true;
        }
    }
}


void WaveManager::startNextWave() {
    if (currentWave >= 3 || !isWaveComplete()) {
        return;
    }

    currentWave++;

    enemiesSpawnedInWave = 0;
    spawnTimer = 0.0f;

    if (currentWave == 2) {
        totalEnemiesInWave = GameConfig::WAVE_2_COUNT;
    }
    else if (currentWave == 3) {
        totalEnemiesInWave = GameConfig::WAVE_3_COUNT;
    }

    activeEnemies.clear();
    dropGenerated.clear();
    enemyProjectiles.clear();
}


bool WaveManager::isWaveComplete() const {
    if(enemiesSpawnedInWave < totalEnemiesInWave) {
        return false;
    }
    for (const auto& enemy : activeEnemies) {
        if (enemy->getState() != EnemyState::DEAD) {
            return false;
        }
    }
    return true;
}

bool WaveManager::spawnEnemy() {
    static std::mt19937 generator{std::random_device{}()}; //random number generator

    std::uniform_int_distribution<int> laneDistribution(
        0, GameConfig::LANE_COUNT - 1
    ); //random lane selection 

    int lane = laneDistribution(generator); //select a random lane

     
    std::unique_ptr<Enemy> enemy;
    EnemyType enemyType = EnemyType::EASY;

   if (currentWave == 2) {
        if (enemiesSpawnedInWave == 0) {
            enemyType = EnemyType::EASY;
        }
        else if (enemiesSpawnedInWave == 1) {
            enemyType = EnemyType::MEDIUM;
        }
        else {
            std::uniform_int_distribution<int> enemyDistribution(0, 1);
            enemyType = static_cast<EnemyType>(enemyDistribution(generator));
        }
    }
    if (currentWave == 3){
        if (enemiesSpawnedInWave == 0) {
            enemyType = EnemyType::EASY;
        }
        else if (enemiesSpawnedInWave == 1) {
            enemyType = EnemyType::MEDIUM;
        }
        else if (enemiesSpawnedInWave == 2) {
            enemyType = EnemyType::HARD;
        }
        else {
            std::uniform_int_distribution<int> enemyDistribution(0, 2);
            enemyType = static_cast<EnemyType>(enemyDistribution(generator));
        }
    }

    static const Grid grid;
    const auto& lanes = grid.getLanes();

    const Lane& selectedLane = lanes[lane];

    glm::vec2 spawnPosition(
        WorldLayout::enemySpawnX(selectedLane),
        WorldLayout::standingY(selectedLane, WorldLayout::ENEMY_HEIGHT)
    );
    
    if (enemyType == EnemyType::EASY) {
    enemy = std::make_unique<EasyEnemy>(lane, spawnPosition, 2.0f);
    }
    else if (enemyType == EnemyType::MEDIUM){
        enemy = std::make_unique<MediumEnemy>(lane, spawnPosition, 2.0f);
    }
    else {
        enemy = std::make_unique<HardEnemy>(lane, spawnPosition, 2.0f);
    }

    activeEnemies.push_back(std::move(enemy));
    dropGenerated.push_back(false);
    return true;
  
}

bool WaveManager::spawnEnemyProjectile(Enemy *enemy) {
    if (enemy == nullptr) {
        return false;
    }
    if(!enemy->canAttack()){
        return false;
    }
    AttackType type = enemy->getAttackType();
    if(type == AttackType::Melee || type == AttackType::Projectile){
        return false;
    }

    static const Grid grid; 
    const auto& lane = grid.getLanes(); 
    int laneIndex = enemy->getLane();

    if(laneIndex < 0 || laneIndex >= 4){
        return false;
    }
    
    const Lane& enemyLane = lane[laneIndex];
    float playerTopY = WorldLayout::standingY(enemyLane, WorldLayout::PLAYER_HEIGHT);
    float projectileCenterY;

    if (type == AttackType::Low){
        projectileCenterY = playerTopY + 
        WorldLayout::PLAYER_HEIGHT * WorldLayout::LOW_ATTACK_OFFSET;
    }
    else if (type == AttackType::High){
        projectileCenterY = playerTopY +
        WorldLayout::PLAYER_HEIGHT * WorldLayout::HIGH_ATTACK_OFFSET;
    }
    else {
        return false;
    }

    float projectileY = projectileCenterY - EnemyProjectile::HEIGHT / 2.0f;

    glm::vec2 position = enemy->getPosition();
    float projectileX =position.x - EnemyProjectile::WIDTH;

    auto projectile = std::make_unique<EnemyProjectile>(
        projectileX,
        projectileY,
        laneIndex,
        0.15f, 
        enemy->getDamage(),
        type
    );
    enemyProjectiles.push_back(std::move(projectile));
    enemy->resetAttackTimer();
    return true;
}
    
void WaveManager::update(float deltaTime) {
    if (deltaTime < 0.0f || gameWon) {
        return;
    }

    // Update existing enemies
    for (auto& enemy : activeEnemies) {
        enemy->update(deltaTime);
        spawnEnemyProjectile(enemy.get()); 
    }
    // Update projectiles
    for(auto& projectile : enemyProjectiles) {
        projectile->update(deltaTime);
        
        if (projectile->getX() + projectile->getWidth() < 0.0f) {
            projectile->setActive(false);
        }
    }
    //delete projectiles that are no longer active
    for(std::size_t i = enemyProjectiles.size(); i > 0; --i) {
        if(!enemyProjectiles[i - 1]->isActive()) {
            enemyProjectiles.erase(enemyProjectiles.begin() + (i - 1));
        }
    }

    // Update spawning timer
    spawnTimer += deltaTime;

    // Spawn enemies
    while (spawnTimer >= GameConfig::ENEMY_SPAWN_INTERVAL &&
           enemiesSpawnedInWave < totalEnemiesInWave) {

        if (!spawnEnemy()) {
            break;
        }

        enemiesSpawnedInWave++;
        spawnTimer -= GameConfig::ENEMY_SPAWN_INTERVAL;
    }
    processEnemyDeaths();
    // Check whether the wave is complete
    if (isWaveComplete()) {
        if (currentWave == 3) {
            gameWon = true;
        }
        else {
            startNextWave();
        }
    }
    
}

             0.60f, 0.35f, 0.06f);

    // Current balance
    std::string balance =
        std::to_string(std::max(0, coins));

    int digitX = x + 64;

    for (char character : balance)
    {
        drawDigit(character - '0',
                  digitX, y + 14);

        digitX += 22;
    }

    // Restore OpenGL state.
    glScissor(
        oldScissor[0], oldScissor[1],
        oldScissor[2], oldScissor[3]);

    glClearColor(
        oldClearColor[0], oldClearColor[1],
        oldClearColor[2], oldClearColor[3]);

    if (!scissorWasEnabled)
    {
        glDisable(GL_SCISSOR_TEST);
    }
}




void HUD::drawhealth(int x,int y,int w, int h,int health,int maxHealth)
{
    if (maxHealth <= 0)
        return;
    if (health < 0)
        health = 0;
    if (health > maxHealth)
        health = maxHealth;
    float actualHealth = (health/(float)maxHealth)*w;
    glEnable(GL_SCISSOR_TEST);
    drawRect(x,y,w,h,0.2f,0.2f,0.2f);

    drawRect(x, y, (int)actualHealth, h, 0.0f, 1.0f, 0.0f);
    glDisable(GL_SCISSOR_TEST);
}

    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);

    glDrawArrays(GL_TRIANGLES, 6, 6);

void PlayerCard:: draw(int hoveredCard)
{
    glUseProgram(shaderProgram);
    glActiveTexture(GL_TEXTURE0);

    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);

    glBindVertexArray(cardVAO);

    int hoveredLocation =glGetUniformLocation(shaderProgram, "hovered");
    glUniform1i(hoveredLocation, hoveredCard == 0);
    glBindTexture(GL_TEXTURE_2D, defenseTexture);
    glDrawArrays(GL_TRIANGLES, 0, 6);

    glUniform1i(hoveredLocation, hoveredCard == 1);
    glBindTexture(GL_TEXTURE_2D, meleeTexture);
    glDrawArrays(GL_TRIANGLES, 6, 6);

    glUniform1i(hoveredLocation, hoveredCard == 2);
    glBindTexture(GL_TEXTURE_2D, rangedTexture);
    glDrawArrays(GL_TRIANGLES, 12, 6);

    glBindVertexArray(0);
    glBindTexture(GL_TEXTURE_2D, 0);

    glDisable(GL_BLEND);
}

#include "ResourceManager.h"

#include <iostream>
#include "stb_image.h"

std::unordered_map<std::string, unsigned int>
    ResourceManager::textures;

unsigned int ResourceManager::loadTexture(const std::string& path)
{
    // Reuse an existing texture if it has already been loaded.
    auto existing = textures.find(path);

    if (existing != textures.end())
    {
        return existing->second;
    }

    int width = 0;
    int height = 0;
    int channels = 0;

    // Force four channels so PNG transparency is preserved.
    stbi_set_flip_vertically_on_load(true);

    unsigned char* pixels = stbi_load(
        path.c_str(),
        &width,
        &height,
        &channels,
        STBI_rgb_alpha
    );

    if (pixels == nullptr)
    {
        std::cerr << "Failed to load texture: " << path << '\n';
        return 0;
    }

    unsigned int texture = 0;

    glGenTextures(1, &texture);
    glBindTexture(GL_TEXTURE_2D, texture);

    glTexImage2D(
        GL_TEXTURE_2D,
        0,
        GL_RGBA,
        width,
        height,
        0,
        GL_RGBA,
        GL_UNSIGNED_BYTE,
        pixels
    );

    glGenerateMipmap(GL_TEXTURE_2D);

    glTexParameteri(
        GL_TEXTURE_2D,
        GL_TEXTURE_WRAP_S,
        GL_CLAMP_TO_EDGE
    );

    glTexParameteri(
        GL_TEXTURE_2D,
        GL_TEXTURE_WRAP_T,
        GL_CLAMP_TO_EDGE
    );

    glTexParameteri(
        GL_TEXTURE_2D,
        GL_TEXTURE_MIN_FILTER,
        GL_LINEAR_MIPMAP_LINEAR
    );

    glTexParameteri(
        GL_TEXTURE_2D,
        GL_TEXTURE_MAG_FILTER,
        GL_LINEAR
    );

    stbi_image_free(pixels);

    textures[path] = texture;

    glBindTexture(GL_TEXTURE_2D, 0);

    std::cout << "Loaded texture: " << path << '\n';

    return texture;
}

void ResourceManager::clear()
{
    for (const auto& entry : textures)
    {
        glDeleteTextures(1, &entry.second);
    }

    textures.clear();
}

        return 1.0f - worldY - objectHeight;

const std::array<Lane, 4>& Grid::getLanes() const {
    return lanes;
}

int Grid::getLaneAtPosition(float x, float y) const {

    for (const Lane& lane : lanes) {

        if (x >= lane.left &&
            x <= lane.right &&
            y >= lane.top &&
            y < lane.bottom) {

            return lane.id;
        }
    }

    return -1;
}
void Grid::toggleLaneSelection(int laneId) {

    if (laneId < 0 || laneId >= 4) {
        return;
    }

    lanes[laneId].selected = !lanes[laneId].selected;
}

    cooldown.update(2.0f);
    check(cooldown.canAttack(), "Cooldown runs during FIGHTING state");
    check(approximatelyEqual(cooldown.getPosition().x, beforeFight),
          "FIGHTING state stops movement");
    cooldown.setState(EnemyState::MOVING);
    cooldown.update(0.25f);
    check(approximatelyEqual(cooldown.getPosition().x, beforeFight - 15.0f),
          "Movement resumes and handles fractional deltaTime");

    std::cout << "\n--- Damage Boundaries and Invalid States ---\n";
    Enemy damageBoundary(7, 2, 110.0f, 1.0f, 1, glm::vec2(900.0f, 175.0f));
    damageBoundary.takeDamage(0);
    check(damageBoundary.getHealth() == 7, "Zero damage is ignored");
    damageBoundary.takeDamage(-1);
    check(damageBoundary.getHealth() == 7, "Negative damage is ignored");
    damageBoundary.takeDamage(6);
    check(damageBoundary.getHealth() == 1 &&
          damageBoundary.getState() != EnemyState::DEAD,
          "Damage leaving exactly 1 HP does not kill");

    check(damageBoundary.getHealth() == 0 &&
          damageBoundary.getState() == EnemyState::DEAD,
          "Exact lethal damage clamps to zero and kills");
    damageBoundary.takeDamage(std::numeric_limits<int>::max());

          "Additional damage to dead enemy is ignored");

    check(damageBoundary.getState() == EnemyState::DEAD,

    damageBoundary.resetAttackTimer();
    check(!damageBoundary.canAttack(), "Dead enemy cannot attack after timer reset");

    Enemy livingState(5, 1, 60.0f, 2.0f, 0, glm::vec2(800.0f, 100.0f));
    livingState.setState(EnemyState::FIGHTING);
    livingState.setState(EnemyState::DEAD);
    check(livingState.getState() == EnemyState::FIGHTING,
          "setState(DEAD) cannot kill from FIGHTING without damage");
    livingState.setState(EnemyState::MOVING);
    check(livingState.getState() == EnemyState::MOVING,
          "Living enemy may return to MOVING");
    livingState.takeDamage(std::numeric_limits<int>::max());
    check(livingState.getHealth() == 0 &&
          livingState.getState() == EnemyState::DEAD,
          "Large overkill damage is clamped to zero");

    std::cout << "\n--- Config, Inheritance, and Ownership ---\n";
    EasyEnemy extraEasy(0, glm::vec2(900.0f, 100.0f), 0.75f);

    HardEnemy extraHard(3, glm::vec2(900.0f, 325.0f), 3.0f);

          extraEasy.getDamage() == GameConfig::EASY_ENEMY_DAMAGE &&

    check(extraMedium.getHealth() == GameConfig::MEDIUM_ENEMY_HP &&
          extraMedium.getDamage() == GameConfig::MEDIUM_ENEMY_DAMAGE &&
          approximatelyEqual(extraMedium.getSpeed(), GameConfig::MEDIUM_ENEMY_SPEED),
          "MediumEnemy initial stats follow GameConfig");
    check(extraHard.getHealth() == GameConfig::HARD_ENEMY_HP &&
          extraHard.getDamage() == GameConfig::HARD_ENEMY_DAMAGE &&

    check(extraEasy.getLane() == 0 && extraMedium.getLane() == 1 &&
          extraHard.getLane() == 3,
          "Subclass constructors preserve valid lane indices");

    std::unique_ptr<Enemy> polymorphic =

    check(dynamic_cast<HardEnemy*>(polymorphic.get()) != nullptr,

    check(approximatelyEqual(polymorphic->getPosition().x,
                             900.0f - GameConfig::HARD_ENEMY_SPEED),
          "Parent-class unique_ptr can update subclass movement");
    polymorphic->takeDamage(GameConfig::HARD_ENEMY_HP);
    check(polymorphic->getState() == EnemyState::DEAD,
          "Parent-class unique_ptr can kill subclass enemy");

    struct DestructionProbe : Enemy {

    check(derivedDestructorCalled,
          "Virtual destructor calls derived destructor through Enemy pointer");

   
std::cout << "\n--- Negative deltaTime Regression ---\n";

Enemy negativeTime(5, 1, 60.0f, 2.0f, 0,
                   glm::vec2(900.0f, 100.0f));

float timerBeforeNegative = negativeTime.getTimeSinceLastAttack();

negativeTime.update(-1.0f);

check(approximatelyEqual(negativeTime.getPosition().x, 900.0f),
      "Negative deltaTime must not move enemy backwards");

check(approximatelyEqual(negativeTime.getTimeSinceLastAttack(),
                         timerBeforeNegative),
      "Negative deltaTime must not reduce attack timer");

    // -----------------------------------------
    // TEST: Virtual Attack Types
    // -----------------------------------------

    std::cout << "\n--- Enemy Attack Type Tests ---\n";

    EasyEnemy attackEasy(0, glm::vec2(900.0f, 100.0f), 2.0f);

    check(easyPtr->getAttackType() == AttackType::High,

    check(mediumPtr->getAttackType() == AttackType::Low,
          "MediumEnemy virtual dispatch works");

    check(hardPtr->getAttackType() == AttackType::Melee,

    std::cout << "\n--- Immediate First Attack Tests ---\n";

    EasyEnemy firstAttackEnemy(

    check(firstAttackEnemy.canAttack(),

    std::cout << "==============================\n";

    if (failed == 0) {

namespace {

int passed = 0;
int failed = 0;

void check(bool condition, const std::string& description) {
    if (condition) {
        ++passed;
        std::cout << "[PASS] " << description << '\n';
    } else {
        ++failed;
        std::cerr << "[FAIL] " << description << '\n';
    }
}

bool near(float a, float b) {
    return std::fabs(a - b) < 0.001f;
}

bool samePosition(const glm::vec2& a, const glm::vec2& b) {
    return near(a.x, b.x) && near(a.y, b.y);
}

bool validLaneAndSpawn(const Enemy* enemy) {
    if (!enemy) return false;
    const int lane = enemy->getLane();
    return lane >= 0 && lane < GameConfig::LANE_COUNT &&
           near(enemy->getPosition().x, 900.0f) &&
           near(enemy->getPosition().y, 100.0f + lane * 75.0f);
}

bool isEasy(const Enemy* e) {
    return dynamic_cast<const EasyEnemy*>(e) != nullptr;
}
bool isMedium(const Enemy* e) {
    return dynamic_cast<const MediumEnemy*>(e) != nullptr;
}
bool isHard(const Enemy* e) {
    return dynamic_cast<const HardEnemy*>(e) != nullptr;
}

bool validTypeForWave(int wave, const Enemy* enemy) {
    if (!enemy) return false;
    if (wave == 1) return isEasy(enemy);
    if (wave == 2) return isEasy(enemy) || isMedium(enemy);
    if (wave == 3) return isEasy(enemy) || isMedium(enemy) || isHard(enemy);
    return false;
}

void defeatAll(WaveManager& manager) {
    // WaveManager owns the enemies; do not erase them while iterating.
    for (std::size_t i = 0; i < manager.getActiveEnemyCount(); ++i) {
        Enemy* enemy = manager.getEnemyAt(i);
        if (enemy) enemy->takeDamage(1000);
    }
}

void checkInitialState() {
    std::cout << "\n--- Initial state / getters ---\n";
    WaveManager manager;
    check(manager.getCurrentWave() == 1, "Starts at Wave 1");
    check(manager.getEnemiesSpawnedInWave() == 0, "Starts with zero spawned enemies");

    check(manager.getEnemyAt(0) == nullptr, "Index 0 is invalid when vector empty");
    check(manager.getEnemyAt(100) == nullptr, "Large index is invalid when vector empty");

    static_assert(
        std::is_same<decltype(std::declval<WaveManager&>().getEnemyAt(0)), Enemy*>::value,
        "Non-const WaveManager must provide mutable Enemy* getter");
    static_assert(
        std::is_same<decltype(std::declval<const WaveManager&>().getEnemyAt(0)), const Enemy*>::value,
        "Const WaveManager must provide const Enemy* getter");
    const WaveManager& readOnly = manager;
    check(readOnly.getEnemyAt(0) == nullptr, "Const getter handles invalid index");

    check(manager.getDropCount() == 0, "New game starts with zero collectibles");
    check(manager.getDropAt(0) == nullptr, "Empty drop list rejects index zero");
    check(manager.getDropAt(9999) == nullptr, "Empty drop list rejects large index");
    check(!manager.removeDropAt(0), "Cannot remove a drop from an empty list");
    check(readOnly.getDropAt(0) == nullptr, "Drop getter works on const WaveManager");
}

void checkTimeBoundaries() {
    std::cout << "\n--- Spawn timing / exact boundaries ---\n";
    WaveManager manager;
    manager.update(0.0f);
    check(manager.getEnemiesSpawnedInWave() == 0, "Zero time does not spawn");
    manager.update(2.5f);
    check(manager.getEnemiesSpawnedInWave() == 0, "2.5 seconds is below spawn interval");
    manager.update(-50.0f);
    check(manager.getEnemiesSpawnedInWave() == 0, "Negative time before spawn is ignored");
    manager.update(0.5f);
    check(manager.getEnemiesSpawnedInWave() == 1, "Exactly 3 seconds spawns first enemy");

    const Enemy* first = manager.getEnemyAt(0);
    check(isEasy(first), "Wave 1's first enemy is EasyEnemy");
    check(validLaneAndSpawn(first), "First enemy spawns in a valid lane and position");
    check(first && near(first->getTimeSinceLastAttack(), first->getAttackInterval()),
          "Newly spawned enemy's attack timer starts at its interval");
    check(first && first->canAttack(), "Newly spawned enemy can attack immediately");
    check(manager.getDropCount() == 0, "Spawning an enemy does not create a drop");

    manager.update(2.5f);
    check(manager.getEnemiesSpawnedInWave() == 1, "Another 2.5 seconds does not spawn early");
    manager.update(0.5f);
    check(manager.getEnemiesSpawnedInWave() == 2, "Second enemy spawns at 6 seconds");
    manager.update(3.0f);
    check(manager.getEnemiesSpawnedInWave() == 3, "Third enemy spawns at 9 seconds");
    manager.update(30.0f);
    check(manager.getEnemiesSpawnedInWave() == 3 && manager.getActiveEnemyCount() == 3,
          "Wave 1 never exceeds three enemies");
    check(manager.getCurrentWave() == 1, "Wave 1 does not advance while enemies are alive");
    check(manager.getEnemyAt(3) == nullptr, "Index exactly equal to size is invalid");
    check(manager.getEnemyAt(9999) == nullptr, "Very large index is invalid");
    check(manager.getDropCount() == 0, "Living enemies do not generate drops over time");

    WaveManager largeFrame;
    largeFrame.update(9.0f);
    check(largeFrame.getActiveEnemyCount() == 3,
          "One 9-second update spawns three enemies");
    bool allStartAtEdge = true;
    for (std::size_t i = 0; i < largeFrame.getActiveEnemyCount(); ++i) {
        allStartAtEdge &= validLaneAndSpawn(largeFrame.getEnemyAt(i));
    }
    check(allStartAtEdge, "All new enemies start at spawn edge in same large frame");
    largeFrame.update(0.0f);
    check(largeFrame.getActiveEnemyCount() == 3,
          "Zero-time update does not duplicate spawns");
}

void checkMovementAndStates() {
    std::cout << "\n--- Movement / negative delta / state ---\n";
    WaveManager manager;
    manager.update(3.0f);
    Enemy* enemy = manager.getEnemyAt(0);
    check(enemy != nullptr, "Can access existing enemy by mutable getter");
    if (!enemy) return;

    // An enemy starts ready for its first attack. Reset to test the next cooldown.
    enemy->resetAttackTimer();
    const float laneY = enemy->getPosition().y;
    manager.update(0.5f);
    check(near(enemy->getPosition().x, 870.0f),
          "EasyEnemy moves left 30 units in 0.5 s");
    check(near(enemy->getPosition().y, laneY), "Movement does not change lane Y");
    check(near(enemy->getTimeSinceLastAttack(), 0.5f),
          "WaveManager updates enemy attack timer");
    enemy->setState(EnemyState::FIGHTING);
    const float beforeFight = enemy->getPosition().x;
    manager.update(1.0f);
    check(near(enemy->getPosition().x, beforeFight), "FIGHTING enemy does not move");
    check(near(enemy->getTimeSinceLastAttack(), 1.5f),
          "FIGHTING enemy continues attack timer");
    enemy->setState(EnemyState::MOVING);
    manager.update(0.5f);
    check(near(enemy->getPosition().x, 840.0f), "MOVING state resumes movement");

    const float beforeNegative = enemy->getPosition().x;
    const float attackTimeBeforeNegative = enemy->getTimeSinceLastAttack();
    const int spawnedBeforeNegative = manager.getEnemiesSpawnedInWave();
    manager.update(-1.0f);
    check(near(enemy->getPosition().x, beforeNegative),
          "Negative WaveManager delta does not move an existing enemy");
    check(near(enemy->getTimeSinceLastAttack(), attackTimeBeforeNegative),
          "Negative WaveManager delta does not change cooldown");
    check(manager.getEnemiesSpawnedInWave() == spawnedBeforeNegative,
          "Negative WaveManager delta does not spawn an enemy");
}

void checkProgressionAndVictory() {
    std::cout << "\n--- Wave transitions, variety, victory ---\n";
    WaveManager manager;
    manager.update(6.0f);
    check(manager.getActiveEnemyCount() == 2 && manager.getCurrentWave() == 1,
          "Wave 1 contains two enemies after 6 seconds");

    defeatAll(manager);
    manager.update(0.0f);
    check(manager.getCurrentWave() == 1 && !manager.hasWon(),
          "Killing all spawned enemies early does not skip remaining spawn quota");
    check(manager.getDropCount() == 2,
          "First two Wave 1 deaths generate two drops before wave completion");

    manager.update(3.0f);
    check(manager.getCurrentWave() == 1 && manager.getActiveEnemyCount() == 3,
          "Final Wave 1 enemy still spawns after earlier kills");
    manager.update(0.0f);
    check(manager.getCurrentWave() == 1, "Wave 1 waits for final living enemy");
    if (Enemy* last = manager.getEnemyAt(2)) last->takeDamage(1000);
    manager.update(0.0f);
    check(manager.getCurrentWave() == 2, "Wave 1 advances to Wave 2 once all are dead");
    check(manager.getDropCount() == 3,
          "All three Wave 1 drops survive the Wave 1-to-2 transition");
    check(manager.getActiveEnemyCount() == 0, "Previous wave enemies are cleared");
    check(manager.getEnemyAt(0) == nullptr, "Old wave indices do not remain valid");
    check(manager.getEnemiesSpawnedInWave() == 0, "Wave 2 spawn counter resets");
    check(!manager.hasWon(), "No premature victory at start of Wave 2");

    manager.update(0.0f);
    check(manager.getActiveEnemyCount() == 0, "No immediate spawn upon transition");
    check(manager.getDropCount() == 3,
          "Zero-time update in Wave 2 preserves previous drops");
    manager.update(2.5f);
    check(manager.getActiveEnemyCount() == 0, "Wave 2 spawn timer starts fresh");
    manager.update(0.5f);
    check(manager.getActiveEnemyCount() == 1 && isEasy(manager.getEnemyAt(0)),
          "First Wave 2 enemy is guaranteed Easy");
    manager.update(12.0f);
    check(manager.getEnemiesSpawnedInWave() == GameConfig::WAVE_2_COUNT &&
          manager.getActiveEnemyCount() == GameConfig::WAVE_2_COUNT,
          "Wave 2 spawns exactly five enemies");
    check(isMedium(manager.getEnemyAt(1)), "Second Wave 2 enemy is guaranteed Medium");

    bool wave2TypesAndLanesOK = true;
    for (std::size_t i = 0; i < manager.getActiveEnemyCount(); ++i) {
        wave2TypesAndLanesOK &= validTypeForWave(2, manager.getEnemyAt(i));
        const Enemy* e = manager.getEnemyAt(i);
        if (!e || e->getLane() < 0 || e->getLane() >= GameConfig::LANE_COUNT ||
            !near(e->getPosition().y, 100.0f + e->getLane() * 75.0f)) {
            wave2TypesAndLanesOK = false;
        }
    }
    check(wave2TypesAndLanesOK, "Every Wave 2 enemy has an allowed type and valid lane");
    check(manager.getDropCount() == 3,
          "Spawning Wave 2 enemies does not duplicate Wave 1 drops");

    // Killing four of five must not advance the wave.
    for (std::size_t i = 0; i < 4; ++i) {
        if (Enemy* e = manager.getEnemyAt(i)) e->takeDamage(1000);
    }
    manager.update(0.0f);
    check(manager.getCurrentWave() == 2 && !manager.hasWon(),
          "Wave 2 does not advance with one enemy still alive");
    check(manager.getDropCount() == 7,
          "Four Wave 2 deaths add exactly four drops");
    if (Enemy* last = manager.getEnemyAt(4)) last->takeDamage(1000);
    manager.update(0.0f);
    check(manager.getCurrentWave() == 3 && manager.getActiveEnemyCount() == 0,
          "Wave 2 advances to Wave 3 only after last death");
    check(manager.getDropCount() == 8,
          "All eight drops from Waves 1 and 2 survive Wave 3 transition");
    check(manager.getEnemiesSpawnedInWave() == 0 && !manager.hasWon(),
          "Wave 3 counters reset and victory is false");

    // Spawn nine of ten enemies; verify all three types before killing them.
    manager.update(27.0f);
    check(manager.getEnemiesSpawnedInWave() == 9 && manager.getActiveEnemyCount() == 9,
          "Nine enemies spawn before Wave 3's tenth spawn interval");
    check(isEasy(manager.getEnemyAt(0)) && isMedium(manager.getEnemyAt(1)) &&
          isHard(manager.getEnemyAt(2)),
          "Wave 3 guarantees Easy, Medium, and Hard as first three enemies");

    bool wave3TypesAndLanesOK = true;
    for (std::size_t i = 0; i < manager.getActiveEnemyCount(); ++i) {
        const Enemy* e = manager.getEnemyAt(i);
        wave3TypesAndLanesOK &= validTypeForWave(3, e);
        if (!e || e->getLane() < 0 || e->getLane() >= GameConfig::LANE_COUNT ||
            !near(e->getPosition().y, 100.0f + e->getLane() * 75.0f)) {
            wave3TypesAndLanesOK = false;
        }
    }
    check(wave3TypesAndLanesOK, "Every Wave 3 enemy has a valid type and lane");

    // First three are guaranteed Easy, Medium, Hard: verify each death drops once.
    const std::size_t dropsBeforeThreeTypes = manager.getDropCount();
    for (std::size_t i = 0; i < 3; ++i) {
        if (Enemy* e = manager.getEnemyAt(i)) e->takeDamage(1000);
    }
    manager.update(0.0f);
    check(manager.getDropCount() == dropsBeforeThreeTypes + 3,
          "Easy, Medium, Hard each generate exactly one collectible");
    manager.update(0.0f);
    check(manager.getDropCount() == dropsBeforeThreeTypes + 3,
          "Repeated updates do not duplicate drops from any enemy type");
    check(manager.getCurrentWave() == 3 && !manager.hasWon(),
          "Killing only first three Wave 3 enemies does not end game");

    defeatAll(manager);
    manager.update(0.0f);
    check(!manager.hasWon() && manager.getCurrentWave() == 3,
          "No victory until all ten Wave 3 enemies have spawned");
    check(manager.getDropCount() == 17,
          "Nine Wave 3 deaths yield 17 cumulative collectibles");
    manager.update(3.0f);
    check(manager.getActiveEnemyCount() == GameConfig::WAVE_3_COUNT &&
          manager.getEnemiesSpawnedInWave() == GameConfig::WAVE_3_COUNT,
          "Tenth Wave 3 enemy spawns at 30 seconds");
    check(!manager.hasWon(), "Spawning tenth living enemy does not trigger victory");
    check(manager.getDropCount() == 17,
          "Tenth living enemy has not generated its drop");

    if (Enemy* last = manager.getEnemyAt(9)) last->takeDamage(1000);
    manager.update(0.0f);
    check(manager.hasWon(), "All spawned and defeated Wave 3 enemies trigger victory");
    check(manager.getDropCount() == 18,
          "Final enemy's collectible is retained when victory occurs");
    check(manager.getDropAt(17) != nullptr,
          "Final victory collectible is accessible by index");
    check(manager.getCurrentWave() == 3, "Game never advances to Wave 4");

    const std::size_t finalCount = manager.getActiveEnemyCount();
    const int finalSpawns = manager.getEnemiesSpawnedInWave();
    manager.update(100.0f);
    manager.update(0.0f);
    check(manager.hasWon() && manager.getCurrentWave() == 3,
          "Victory remains true after future updates");
    check(manager.getActiveEnemyCount() == finalCount &&
          manager.getEnemiesSpawnedInWave() == finalSpawns,
          "No extra enemies spawn after victory");
    check(manager.getDropCount() == 18,
          "Updates after victory cannot duplicate any of the 18 drops");
}

    Enemy* enemy = manager.getEnemyAt(0);
    if (!enemy) {
        check(false, "Spawned enemy is accessible for drop tests");

    }

    const glm::vec2 expectedPosition = enemy->getPosition();
    const int expectedLane = enemy->getLane();
    enemy->takeDamage(1000);
    check(manager.getDropCount() == 0,
          "Death is processed by WaveManager.update, not immediately by takeDamage");
    manager.update(0.0f);
    check(manager.getDropCount() == 1, "Dead enemy generates one collectible");

    const EnemyDrop* drop = manager.getDropAt(0);
    check(drop && samePosition(drop->position, expectedPosition) &&
          drop->lane == expectedLane,
          "Collectible appears at dead enemy's position and lane");
    check(manager.getDropAt(1) == nullptr,
          "Index equal to drop count returns nullptr");
    check(manager.getDropAt(10) == nullptr,
          "Invalid collectible index returns nullptr");

    manager.update(0.0f);
    manager.update(0.0f);
    manager.update(0.5f);
    check(manager.getDropCount() == 1,
          "Repeated updates do not duplicate collectibles");

    const std::size_t enormous = std::numeric_limits<std::size_t>::max();
    check(manager.getDropAt(enormous) == nullptr,
          "Very large collectible index returns nullptr");
    check(!manager.removeDropAt(10), "Cannot remove nonexistent collectible");
    check(!manager.removeDropAt(enormous), "Very large removal index is rejected");
    check(manager.getDropCount() == 1,
          "Invalid removal does not change drop count");
    check(manager.removeDropAt(0), "Collectible is removed successfully");
    check(manager.getDropCount() == 0, "No collectibles remain after pickup");
    check(manager.getDropAt(0) == nullptr,
          "Collected drop cannot be accessed after removal");
    check(!manager.removeDropAt(0), "Same collectible cannot be collected twice");
    manager.update(0.0f);
    check(manager.getDropCount() == 0,
          "Already-collected drop is not regenerated by a dead enemy");
}

void checkMultipleDropsAndRemovalOrder() {
    std::cout << "\n--- Multiple deaths / removal order ---\n";
    WaveManager manager;
    manager.update(9.0f); // Wave 1: all three EasyEnemies spawn.
    check(manager.getActiveEnemyCount() == 3,
          "Three enemies available for multiple-drop test");
    if (manager.getActiveEnemyCount() != 3) return;

    const Enemy* first = manager.getEnemyAt(0);
    const Enemy* second = manager.getEnemyAt(1);
    const Enemy* third = manager.getEnemyAt(2);
    if (!first || !second || !third) {
        check(false, "All three enemies accessible for removal-order test");
        return;
    }
    const glm::vec2 firstPosition = first->getPosition();
    const glm::vec2 secondPosition = second->getPosition();
    const glm::vec2 thirdPosition = third->getPosition();
    const int firstLane = first->getLane();
    const int secondLane = second->getLane();
    const int thirdLane = third->getLane();

    // Process deaths one at a time so drop order is deterministic.
    manager.getEnemyAt(0)->takeDamage(1000);
    manager.update(0.0f);
    check(manager.getDropCount() == 1,
          "First enemy death produces first drop");
    check(manager.getCurrentWave() == 1,
          "Wave remains active while two enemies are alive");

    manager.getEnemyAt(1)->takeDamage(1000);
    manager.update(0.0f);
    check(manager.getDropCount() == 2,
          "Second enemy death produces second drop");

    manager.getEnemyAt(2)->takeDamage(1000);
    manager.update(0.0f);
    check(manager.getCurrentWave() == 2,
          "Wave advances when last of three enemies dies");
    check(manager.getDropCount() == 3,
          "Last enemy's drop survives when active enemies are cleared");

    const EnemyDrop* drop0 = manager.getDropAt(0);
    const EnemyDrop* drop1 = manager.getDropAt(1);
    const EnemyDrop* drop2 = manager.getDropAt(2);
    check(drop0 && samePosition(drop0->position, firstPosition) && drop0->lane == firstLane,
          "First drop preserves first enemy position and lane");
    check(drop1 && samePosition(drop1->position, secondPosition) && drop1->lane == secondLane,
          "Second drop preserves second enemy position and lane");
    check(drop2 && samePosition(drop2->position, thirdPosition) && drop2->lane == thirdLane,
          "Third drop preserves third enemy position and lane");

    check(manager.removeDropAt(1), "Removing middle drop succeeds");
    check(manager.getDropCount() == 2, "Middle removal reduces drop count to two");
    const EnemyDrop* newFirst = manager.getDropAt(0);
    const EnemyDrop* newSecond = manager.getDropAt(1);
    check(newFirst && samePosition(newFirst->position, firstPosition) &&
          newFirst->lane == firstLane,
          "Middle removal preserves the earlier drop");
    check(newSecond && samePosition(newSecond->position, thirdPosition) &&
          newSecond->lane == thirdLane,
          "Middle removal shifts the later drop into the removed index");
    check(manager.removeDropAt(1), "Removing last remaining index succeeds");
    check(manager.removeDropAt(0), "Removing first remaining index succeeds");
    check(manager.getDropCount() == 0, "Removing all drops leaves an empty list");
    manager.update(0.0f);
    check(manager.getDropCount() == 0,
          "Previously collected drops stay removed across a wave transition");
}

void checkNegativeTimeDefersDropProcessing() {
    std::cout << "\n--- Death processing boundaries ---\n";
    WaveManager manager;
    manager.update(3.0f);
    Enemy* enemy = manager.getEnemyAt(0);
    if (!enemy) {
        check(false, "Enemy available for negative-time drop test");
        return;
    }
    enemy->takeDamage(1000);
    manager.update(-1.0f);
    check(manager.getDropCount() == 0,
          "Negative WaveManager update does not process a pending death");
    manager.update(0.0f);
    check(manager.getDropCount() == 1,
          "Zero-time update processes a previously pending death");
    manager.update(0.0f);
    check(manager.getDropCount() == 1,
          "Pending death is recorded only once");
}

void checkDropIndependenceBetweenManagers() {
    std::cout << "\n--- Independent WaveManager drop storage ---\n";
    WaveManager managerA;
    WaveManager managerB;
    managerA.update(3.0f);
    managerB.update(3.0f);
    if (Enemy* e = managerA.getEnemyAt(0)) e->takeDamage(1000);
    managerA.update(0.0f);
    check(managerA.getDropCount() == 1,
          "First WaveManager records its enemy's drop");
    check(managerB.getDropCount() == 0,
          "Separate WaveManager has no drops from first manager");
    check(managerA.removeDropAt(0),
          "First WaveManager can remove its own drop");
    check(managerB.getDropCount() == 0,
          "Removing drops in first WaveManager does not affect another");
}

    checkProgressionAndVictory();
    checkRepeatedFreshManagers();
    checkSingleDropLifecycle();

              << failed << " failed\n";

}
